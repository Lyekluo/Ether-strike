## 1. 动态数组 (Dynamic Array)

**物理意义**：连续的内存块。C++ `std::vector` 的雏形。

```c
typedef struct {
    int *data;
    int size;       // 当前元素个数
    int capacity;   // 当前分配的总容量
} ArrayList;

void initList(ArrayList *list) {
    list->capacity = 4;
    list->size = 0;
    list->data = malloc(sizeof(int) * list->capacity);
}

void pushBack(ArrayList *list, int value) {
    if (list->size == list->capacity) {
        // 扩容翻倍，realloc 会自动拷贝老数据
        list->capacity *= 2;
        int *new_data = realloc(list->data, sizeof(int) * list->capacity);
        if (new_data == NULL){
            return; // 内存分配失败
        }
        list->data = new_data;
    }
    list->data[list->size] = value;
    list->size++;
}
```

> [!WARNING] 碳基防坑
> `realloc` 失败时返回 `NULL`，原指针依然有效。绝对不能写 `list->data = realloc(list->data, ...)`，否则原数据内存直接泄漏！

## 2. 单链表 (Singly Linked List)

**物理意义**：散落的内存节点通过指针串联。重点在于**二级指针 `Node **` 的使用**，用于直接修改头节点。

```c
typedef struct Node {
    int val;
    struct Node *next;
} Node;

// 头插法 (O(1))
void insertFront(Node **head, int value) {
    Node *new_node = malloc(sizeof(Node));
    new_node->val = value;
    new_node->next = *head;
    *head = new_node;
}

// 尾插法 (O(N))
void insertBack(Node **head, int value) {
    Node *new_node = malloc(sizeof(Node));
    new_node->val = value;
    new_node->next = NULL;
    
    if (*head == NULL) {
        *head = new_node;
        return;
    }
    Node *curr = *head;
    while (curr->next != NULL) {
        curr = curr->next;
    }
    curr->next = new_node;
}

// 删除值为 target 的所有节点
void deleteNodes(Node **head, int target) {
    Node *curr = *head;
    Node *prev = NULL;
    while (curr != NULL) {
        if (curr->val == target) {
            Node *to_delete = curr;
            if (prev == NULL) { // 删头节点
                *head = curr->next;
            }
            else {
                prev->next = curr->next;
            }
            curr = curr->next;
            free(to_delete);
        }
        else {
            prev = curr;
            curr = curr->next;
        }
    }
}

// 反转链表 (三指针法)
Node *reverseList(Node *head) {
    Node *prev = NULL;
    Node *curr = head;
    while (curr != NULL) {
        Node *next_node = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next_node;
    }
    return prev;
}
```

## 3. 双向链表 (Doubly Linked List)

**物理意义**：每个节点既有指向前驱的 `prev`，也有指向后继的 `next`。结合 Dummy 节点可消除所有边界特判。

```c
typedef struct DNode {
    int val;
    struct DNode *prev;
    struct DNode *next;
} DNode;

// 初始化带 Dummy 的空链表
DNode *createDoublyList() {
    DNode *head = malloc(sizeof(DNode));
    DNode *tail = malloc(sizeof(DNode));
    head->next = tail;
    head->prev = NULL;
    tail->prev = head;
    tail->next = NULL;
    return head;
}

// 在节点 p 之后插入新节点
void insertAfter(DNode *p, int value) {
    DNode *new_node = malloc(sizeof(DNode));
    new_node->val = value;
    
    new_node->prev = p;
    new_node->next = p->next;
    p->next->prev = new_node;
    p->next = new_node;
}

// 删除节点 p
void deleteNode(DNode *p) {
    if (p->prev == NULL || p->next == NULL){
        return; // 不能删 dummy
    }
    p->prev->next = p->next;
    p->next->prev = p->prev;
    free(p);
}
```

## 4. 栈 (Stack)

**物理意义**：后进先出 (LIFO)。这里用数组实现，比链表实现缓存命中率更高。

```c
#define MAX_STACK 100

typedef struct {
    int data[MAX_STACK];
    int top;
} Stack;

void initStack(Stack *s) {
    s->top = -1;
}

void push(Stack *s, int value) {
    if (s->top < MAX_STACK - 1) {
        s->data[++(s->top)] = value;
    }
}

int pop(Stack *s) {
    if (s->top == -1){
        return -1;
    }
    return s->data[(s->top)--];
}
```

## 5. 队列与循环队列 (Queue)

**物理意义**：先进先出 (FIFO)。普通数组队列出队会导致 \(O(N)\) 的数据搬运，因此必须使用**循环队列**。

```c
#define MAX_QUEUE 100

typedef struct {
    int data[MAX_QUEUE];
    int head;
    int tail;
    int size;
} CircularQueue;

void initQueue(CircularQueue *q) {
    q->head = 0;
    q->tail = 0;
    q->size = 0;
}

void enqueue(CircularQueue *q, int value) {
    if (q->size < MAX_QUEUE) {
        q->data[q->tail] = value;
        q->tail = (q->tail + 1) % MAX_QUEUE; // 取模实现循环
        q->size++;
    }
}

int dequeue(CircularQueue *q) {
    if (q->size == 0){
        return -1;
    }
    int value = q->data[q->head];
    q->head = (q->head + 1) % MAX_QUEUE;
    q->size--;
    return value;
}
```

## 6. 二叉树与遍历 (Binary Tree)

**物理意义**：分形结构。递归是处理树的最自然方式。

```c
typedef struct TreeNode {
    int val;
    struct TreeNode *left;
    struct TreeNode *right;
} TreeNode;

// 前序遍历 (根-左-右)
void preOrder(TreeNode *root) {
    if (root == NULL){
        return;
    }
    printf("%d ", root->val);
    preOrder(root->left);
    preOrder(root->right);
}

// 中序遍历 (左-根-右)
void inOrder(TreeNode *root) {
    if (root == NULL){
        return;
    }
    inOrder(root->left);
    printf("%d ", root->val);
    inOrder(root->right);
}

// 层序遍历 (BFS，基于队列)
void levelOrder(TreeNode *root) {
    if (root == NULL){
        return;
    }
    TreeNode *queue[1000];
    int head = 0, tail = 0;
    queue[tail++] = root;
    
    while (head < tail) {
        TreeNode *curr = queue[head++];
        printf("%d ", curr->val);
        if (curr->left != NULL){
            queue[tail++] = curr->left;
        }
        if (curr->right != NULL){
            queue[tail++] = curr->right;
        }
    }
}
```

## 7. 二叉搜索树 (BST)

**物理意义**：左小右大。增删查平均 \(O(\log N)\)，但如果插入有序数据会退化为链表 \(O(N)\)。

```c
// 插入节点
TreeNode *insertBST(TreeNode *root, int value) {
    if (root == NULL) {
        TreeNode *node = malloc(sizeof(TreeNode));
        node->val = value;
        node->left = NULL;
        node->right = NULL;
        return node;
    }
    if (value < root->val) {
        root->left = insertBST(root->left, value);
    }
    else if (value > root->val) {
        root->right = insertBST(root->right, value);
    }
    return root;
}

// 查找节点
TreeNode *searchBST(TreeNode *root, int target) {
    while (root != NULL && root->val != target) {
        if (target < root->val){
            root = root->left;
        }
        else{
            root = root->right;
        }
    }
    return root;
}
```

## 8. 堆 / 优先队列 (Heap)

**物理意义**：完全二叉树。用数组存储（下标 `i` 的左右孩子是 `2i+1` 和 `2i+2`）。这里实现小根堆。

```c
typedef struct {
    int data[1000];
    int size;
} MinHeap;

void pushHeap(MinHeap *h, int value) {
    h->data[h->size] = value;
    int curr = h->size;
    h->size++;
    
    // 上浮操作
    while (curr > 0) {
        int parent = (curr - 1) / 2;
        if (h->data[parent] <= h->data[curr]){
            break;
        }
        int tmp = h->data[parent];
        h->data[parent] = h->data[curr];
        h->data[curr] = tmp;
        curr = parent;
    }
}

int popHeap(MinHeap *h) {
    if (h->size == 0){
        return -1;
    }
    int min_val = h->data[0];
    h->data[0] = h->data[h->size - 1];
    h->size--;
    
    // 下沉操作
    int curr = 0;
    while (2 * curr + 1 < h->size) {
        int left = 2 * curr + 1;
        int right = 2 * curr + 2;
        int smallest = curr;
        
        if (h->data[left] < h->data[smallest]){
            smallest = left;
        }
        if (right < h->size && h->data[right] < h->data[smallest]){
            smallest = right;
        }
        if (smallest == curr){
            break;
        }
        
        int tmp = h->data[curr];
        h->data[curr] = h->data[smallest];
        h->data[smallest] = tmp;
        curr = smallest;
    }
    return min_val;
}
```

## 9. 哈希表 (拉链法)

**物理意义**：数组 + 链表。通过哈希函数快速定位桶，解决冲突。

```c
#define TABLE_SIZE 100003 // 大质数防冲突

typedef struct HashNode {
    int key;
    int value;
    struct HashNode *next;
} HashNode;

HashNode *hashTable[TABLE_SIZE];

int hashFunc(int key) {
    return key % TABLE_SIZE;
}

void put(int key, int value) {
    int idx = hashFunc(key);
    HashNode *curr = hashTable[idx];
    
    // 查找是否已存在，存在则更新
    while (curr != NULL) {
        if (curr->key == key) {
            curr->value = value;
            return;
        }
        curr = curr->next;
    }
    
    // 头插法插入
    HashNode *new_node = malloc(sizeof(HashNode));
    new_node->key = key;
    new_node->value = value;
    new_node->next = hashTable[idx];
    hashTable[idx] = new_node;
}

int get(int key) {
    int idx = hashFunc(key);
    HashNode *curr = hashTable[idx];
    while (curr != NULL) {
        if (curr->key == key) {
            return curr->value;
        }
        curr = curr->next;
    }
    return -1;
}
```

## 10. 图 (邻接表)

**物理意义**：用数组存顶点，用链表存边。比邻接矩阵更省空间，适合稀疏图。

```c
#define MAX_VERTICES 100

// 边节点
typedef struct EdgeNode {
    int to;
    int weight;
    struct EdgeNode *next;
} EdgeNode;

// 顶点
typedef struct {
    EdgeNode *head;
} Vertex;

Vertex graph[MAX_VERTICES];

// 添加有向边 u -> v
void addEdge(int u, int v, int weight) {
    EdgeNode *new_edge = malloc(sizeof(EdgeNode));
    new_edge->to = v;
    new_edge->weight = weight;
    new_edge->next = graph[u].head;
    graph[u].head = new_edge;
}

// 遍历某个节点的所有邻居
void printNeighbors(int u) {
    EdgeNode *curr = graph[u].head;
    while (curr != NULL) {
        printf("节点 %d 连向 %d, 权重 %d\n", u, curr->to, curr->weight);
        curr = curr->next;
    }
}
```
